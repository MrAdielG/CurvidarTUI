#!/data/data/com.termux/files/usr/bin/python
from babel.dates import format_date, format_time
from textual.app import App
from textual.widgets import Button, ProgressBar, Static, Label
from textual.containers import Center
from datetime import datetime as dt
from pathlib import Path
import sys
import os
import asyncio
from asyncio import sleep

# # --- Script Configuration ---
# # YouTube URL from the share menu
url=sys.argv[1]

# # Download directory in shared storage
ytdir= Path.home() / "storage/shared/YouTube"

# # Output filename template: Limits title to a safe length (e.g., 180 chars)
output_template="%(title).180s.%(ext)s"

# # Ensure the download directory exists
# # Cria o diretório para download
Path(ytdir).mkdir(exist_ok=True)

# # --- Banner Function ---
# # This banner will be shown every time the screen is cleared.

 ██████  ██████  ██   ██   ████   ██ ██████  ██   ██ ██   ██ TUI\n""" + f"""

[not bold red]{format_date(dt.today(), "EEEE, d  'de' MMMM yyyy", locale="pt_BR").title()} | {format_time(dt.now(), format="HH:mm")}"""


# # --- Main Menu Function ---
class ShowMenu(App):
    CSS_PATH = "termux-file-opener.tcss"
    def compose(self):
        yield Banner()
        yield Label("--- Curvidar: YouTube Downloader ---")
        yield Label(f"[yellow]URL: [not bold]{url}")
        yield Button("Baixar Video - 1080p (Qualidade Alta)", classes='video', id='video1080')
        yield Button("Baixar Video - 720p (Qualidade Média)", classes='video', id='video720')
        yield Button("Baixar Video - 480p (Qualidade Padrão)", classes='video', id='video480')
        yield Button("Baixar Apenas Audio (m4a - qualidade)", classes='audio', id='audiom4a')
        yield Button("Baixar Apenas Audio (mp3 - compatibilidade)", classes='audio', id='audiomp3')
        yield Button("SAIR", id='sair')
        with Center():
            yield Static("Aguardando download...", id="status")

    def on_mount(self):
        self.query_one(Center).display = False
    

    async def download(self, choice):
        status = self.query_one("#status", Static)

        formatos = {
            "video1080": ["-f", "bestvideo[height<=1080]+bestaudio/best"],
            "video720": ["-f", "bestvideo[height<=720]+bestaudio/best"],
            "video480": ["-f", "bestvideo[height<=480]+bestaudio/best"],
            "audiom4a": ["-f", "bestaudio[ext=m4a]/bestaudio"],
            "audiomp3": ["-x", "--audio-format", "mp3"]
        }

        argumentos = [
            "yt-dlp",
            "--restrict-filenames",
            "--newline",
            "--progress-template",
            "download:%(progress.percent)s|%(progress._speed_str)s|%(progress._eta_str)s",
            *formatos[choice],
            "-o", output_template,
            url
        ]

        status.display = True
        status.update("Iniciando download...")

        try:
            processo = await asyncio.create_subprocess_exec(
                *argumentos,
                cwd=ytdir,
                stdout=asyncio.subprocess.PIPE,
                stderr=asyncio.subprocess.STDOUT
            )

            while True:
                linha = await processo.stdout.readline()

                if not linha:
                    break

                texto = linha.decode(errors="replace").strip()
                status.update(texto)

            codigo = await processo.wait()

            if codigo == 0:
                status.update("[green]Download concluído!\nVerifique a pasta Youtube.[/green]")
                sleep(5)
                self.exit()
            else:
                status.update(f"[red]Erro no download: {codigo}[/red]")

        except Exception as erro:
            status.update(f"[red]Erro: {erro}[/red]")

    # # Process the user's choice
    async def on_button_pressed(self, event: Button.Pressed):
        if event.button.id == "sair":
            ## TODO: chamar intent do youtube
            self.exit()
            return
        else:
            self.query_one(Center).display = True
            os.chdir(ytdir)
            await self.download(choice=event.button.id)
            
ShowMenu().run()

# # --- Script Configuration ---
# # YouTube URL from the share menu
url=sys.argv[1]

# # Download directory in shared storage
ytdir= Path.home() / "storage/shared/YouTube"

# # Output filename template: Limits title to a safe length (e.g., 180 chars)
output_template="%(title).180s.%(ext)s"

# # Ensure the download directory exists
# # Cria o diretório para download
Path(ytdir).mkdir(exist_ok=True)

# # --- Main Menu Function ---
class ShowMenu(App):
    CSS = """
    ShowMenu {
        Button {
            width: 1fr;
            height: 1fr;
            text-style: bold;
        }

        .video {
            color: green;
            background: transparent;
        }

        #video1080, #video720, #video480 {
            background: $surface;
            margin-bottom: 1;
            border: tall 60%;
        }

        .audio {
            color: rgb(106, 50, 255);
            border: round rgb(106, 50, 255);
            background: transparent;
        }

        #audiom4a, #audiomp3 {
            background: $surface;
            margin-bottom: 1;
            border: tall rgb(106, 50, 255);
        }

        #sair {
            background: $surface;
            color: red;
            border: tall red;
        }


        Screen {
            align: center middle;

            Center {
                layer: above;
                width: 80%;
                height: 65%;
                content-align: center middle;
                align: center middle;
                border: tall blue;
                offset-y: 4;
            }
        }
    }
    """
    def compose(self):
        gradient = Gradient.from_colors(
            "#881177",
            "#aa3355",
            "#cc6666",
            "#ee9944",
            "#eedd00",
            "#99dd55",
            "#44dd88",
            "#22ccbb",
            "#00bbcc",
            "#0099cc",
            "#3366bb",
            "#663399",
        )
        yield Banner()
        yield Label("[cyan]--- Curvidar: YouTube Downloader ---")
        yield Label(f"[yellow]URL: [not bold]{url}")
        yield Button("[bold]Baixar Video - 1080p (Qualidade Alta)", classes='video', id='video1080')
        yield Button("[bold]Baixar Video - 720p (Qualidade Média)", classes='video', id='video720')
        yield Button("[bold]Baixar Video - 480p (Qualidade Padrão)", classes='video', id='video480')
        yield Button("[bold]Baixar Apenas Audio (m4a - qualidade)", classes='audio', id='audiom4a')
        yield Button("[bold]Baixar Apenas Audio (mp3 - compatibilidade)", classes='audio', id='audiomp3')
        yield Button("[bold]SAIR", id='sair')
        with Center():
            yield ProgressBar(total=100, id="progresso", show_eta=False, gradient=gradient, show_percentage=True)
            yield Static("Aguardando download...", id="status")

    def on_mount(self):
        self.download_worker = None
        self.processo = None
        self.query_one(Center).display = False
    

    async def download(self, choice):
        barra = self.query_one("#progresso", ProgressBar)
        barra.styles.padding = (0, 0, 0, 2)
        status = self.query_one("#status", Static)
        status.styles.padding = (0, 0, 0, 2)

        formatos = {
            "video1080": ["-f", "bestvideo[height<=1080]+bestaudio/best"],
            "video720": ["-f", "bestvideo[height<=720]+bestaudio/best"],
            "video480": ["-f", "bestvideo[height<=480]+bestaudio/best"],
            "audiom4a": ["-f", "bestaudio[ext=m4a]/bestaudio"],
            "audiomp3": ["-x", "--audio-format", "mp3"]
        }

        argumentos = [
            "yt-dlp",
            "--restrict-filenames",
            "--downloader", "aria2c",
            "--downloader-args", "aria2c:-x 16 -k 1M",
            *formatos[choice],
            "-o", output_template,
            url
        ]
        barra.update(progress=0)
        status.update("Iniciando download...")

        try:
            self.processo = await asyncio.create_subprocess_exec(
                *argumentos,
                cwd=ytdir,
                stdout=asyncio.subprocess.PIPE,
                stderr=asyncio.subprocess.STDOUT
            )
            buffer = ""
            
            while True:
                
                bloco = await self.processo.stdout.read(1024)
                
                if not bloco:
                    
                    break

                buffer += bloco.decode(errors="replace")

                # aria2c atualiza usando \r, não apenas \n
                partes = re.split(r"[\r\n]+", buffer)

                buffer = partes.pop()

                for texto in partes:
                    resultado = re.search(
                        r"\(([\d.]+)%\).*?DL:([\d.]+)([KMGT]?i?B).*?ETA:([\w:-]+)",
                        texto
                    )

                    if not resultado:
                        status.update("Baixando...")
                        barra.advance(1)
                        continue

                    porcentagem = resultado.groups()[0]

                    progresso = float(porcentagem)

                    diferenca = int(progresso) - int(barra.progress)

                    if diferenca > 0:
                        for _ in range(diferenca):
                            barra.advance(1)
                            await sleep(0.01)

            codigo = await self.processo.wait()

            if codigo == 0:
                diferenca = 100-int(barra.progress)
                if diferenca > 0:
                    for _ in range(diferenca):
                        barra.advance(1)
                        await sleep(0.01)
                barra.update(progress=100)
                status.update("[green]Download concluído!\nVerifique a pasta Youtube.[/green]")
                self.exit()
            else:
                status.update(f"[red]Erro no download: {codigo}[/red]")
        except asyncio.CancelledError:

            if self.processo and self.processo.returncode is None:
                self.processo.terminate()
                await self.processo.wait()

            raise
        except Exception as erro:
            status.update(f"[red]Erro: {erro}[/red]")
        finally:
            self.processo = None

    # # Process the user's choice
    def on_button_pressed(self, event: Button.Pressed):
        if event.button.id == "sair":
            if self.processo and self.processo.returncode is None:
                self.processo.terminate()

            if self.download_worker and self.download_worker.is_running:
                self.download_worker.cancel()
            self.exit()
            return
        
        if self.download_worker and self.download_worker.is_running:
            return

        self.query_one(Center).display = True

        self.download_worker = self.run_worker(
            self.download(event.button.id),
            exclusive=True,
            group="download"
        )
                
            


ShowMenu().run()
